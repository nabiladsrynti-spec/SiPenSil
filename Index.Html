Koneksi Google Spreadsheet (AppScript Web App)

Hubungkan web app ini ke Google Sheets agar setiap data pengawasan otomatis tersimpan di cloud secara online.

Jika URL belum diisi, aplikasi akan menggunakan mode Penyimpanan Lokal (Offline LocalStorage).

Panduan Cara Menghubungkan ke Google Sheets (5 Menit)

  1. Buat Spreadsheet baru di Google Sheets dengan nama "Pengawasan PSAT Ambon".
  2. Buat Header Kolom pada baris ke-1 spreadsheet dengan urutan nama:
    ID | Tanggal | Kategori | Lokasi | Kecamatan | Pedagang | Komoditas | NoIzin | ParameterUji | HasilUji | Status | Tindakan | Petugas | Catatan | WaktuSimpan
  3. Klik menu Ekstensi (Extensions) → Apps Script pada Google Sheet tersebut.
  4. Hapus semua kode bawaan, lalu Copy & Paste kode script berikut ke dalam editor Google Apps Script:
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  var data = [];
  
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    data.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      data.id,
      data.kategori || 'Gudang',
      data.petugas,
      data.waktu,
      data.lokasi || '-',
      data.kecamatan || '-',
      data.jenisPangan || data.jenisKomoditas || data.jenisPsah || '-',
      data.namaDagang || data.namaKomoditas || data.namaPsah || '-',
      data.produsenAsal || data.asalKomoditas || data.asalPsah || '-',
      data.noIzin || '-',
      data.hasilUji || '-',
      data.status,
      data.catatan || '-',
      new Date().toLocaleString('id-ID')
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
